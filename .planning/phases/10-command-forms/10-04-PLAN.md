/**
 * State reader for command form prepopulation
 * Reads STATE.md fresh each time (not cached) per CONTEXT.md decision
 */

import { invoke } from '@tauri-apps/api/core';

interface CurrentState {
  currentPhase: number | null;
}

/**
 * Read current phase number from STATE.md
 * Returns null if STATE.md doesn't exist or can't be parsed
 */
export async function readCurrentPhase(projectPath: string): Promise<number | null> {
  try {
    const statePath = `${projectPath}/.planning/STATE.md`;
    const content = await invoke<string>('read_file', { path: statePath });

    // Parse current phase from STATE.md content
    // Look for "Phase: X of Y" or "Phase X" pattern
    const phaseMatch = content.match(/Phase:\s*(\d+)\s*of\s*\d+/i)
      || content.match(/Phase\s*(\d+)/i);

    if (phaseMatch) {
      return parseInt(phaseMatch[1], 10);
    }

    return null;
  } catch (error) {
    // STATE.md doesn't exist or read failed
    console.warn('Could not read STATE.md for prepopulation:', error);
    return null;
  }
}

/**
 * Read current state values for form prepopulation
 */
export async function readCurrentState(projectPath: string): Promise<CurrentState> {
  const currentPhase = await readCurrentPhase(projectPath);
  return { currentPhase };
}
```

Note: Uses existing Tauri 'read_file' invoke command that's already used elsewhere in the app.

useEffect(() => {
  if (!commandDialogOpen || !selectedCommand || !projectPath) return;

  async function prepopulateForm() {
    // Build default values for parameters
    const paramDefaults: Record<string, string | number | boolean> = {};

    // Check if command has a phase parameter
    const hasPhaseParam = selectedCommand.parameters.some(p => p.name === 'phase');

    if (hasPhaseParam) {
      // Read current phase from STATE.md (fresh read, not cached)
      const currentPhase = await readCurrentPhase(projectPath);
      if (currentPhase !== null) {
        paramDefaults.phase = currentPhase;
      }
    }

    // Add other parameter defaults
    selectedCommand.parameters.forEach(param => {
      if (param.defaultValue !== undefined && paramDefaults[param.name] === undefined) {
        paramDefaults[param.name] = param.defaultValue;
      }
    });

    // Add flag defaults (all false)
    const flagDefaults = (selectedCommand.flags || []).reduce((acc, flag) => {
      acc[flag.name] = false;
      return acc;
    }, {} as Record<string, boolean>);

    // Reset form with prepopulated values
    reset({ ...paramDefaults, ...flagDefaults });
  }

  prepopulateForm();
}, [commandDialogOpen, selectedCommand, projectPath, reset]);
```
