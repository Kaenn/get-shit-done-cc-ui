Create Unix wrapper script (npm/bin/get-shit-done-cc-ui):

```bash
#!/bin/sh
SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
BINARY="$SCRIPT_DIR/gsd-ui-web"

# Check if binary exists
if [ ! -f "$BINARY" ]; then
  echo "Binary not found. Downloading for your platform..."
  # Run download script via node
  node "$SCRIPT_DIR/../lib/download.js"
  DOWNLOAD_EXIT=$?
  if [ $DOWNLOAD_EXIT -ne 0 ]; then
    exit $DOWNLOAD_EXIT
  fi
  echo ""
  echo "Download complete! Run 'npx get-shit-done-cc-ui' again to start."
  exit 0
fi

exec "$BINARY" "$@"
```

Create Windows wrapper script (npm/bin/get-shit-done-cc-ui.cmd):

```batch
@echo off
setlocal
set "SCRIPT_DIR=%~dp0"
set "BINARY=%SCRIPT_DIR%gsd-ui-web.exe"

if not exist "%BINARY%" (
  echo Binary not found. Downloading for your platform...
  node "%SCRIPT_DIR%..\lib\download.js"
  if errorlevel 1 exit /b 1
  echo.
  echo Download complete! Run 'npx get-shit-done-cc-ui' again to start.
  exit /b 0
)

"%BINARY%" %*
```

Key behavior per CONTEXT.md:
- First run: check for binary, if missing -> download -> print success message -> exit
- Second run: binary exists -> exec it with all args
- User must run twice: once to download, once to execute

Make Unix script executable: chmod +x npm/bin/get-shit-done-cc-ui
