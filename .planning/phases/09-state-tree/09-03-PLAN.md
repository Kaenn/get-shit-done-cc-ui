/**
 * Collapsible section for archived milestones
 * Shows past milestones with dimmed styling, collapsed by default
 */

import { useState } from 'react';
import { ChevronRight } from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';
import { cn } from '@/lib/utils';
import { GSDTreeNode } from './GSDTreeNode';
import type { TreeNode } from '@/lib/gsd/tree-transforms';

interface GSDArchivedSectionProps {
  archivedNodes: TreeNode[];
  projectPath: string | null;
}

export function GSDArchivedSection({ archivedNodes, projectPath }: GSDArchivedSectionProps) {
  const [isExpanded, setIsExpanded] = useState(false);

  if (archivedNodes.length === 0) {
    return null;
  }

  return (
    <div className="mt-4 border-t border-border pt-4">
      <button
        onClick={() => setIsExpanded(!isExpanded)}
        className={cn(
          'flex items-center gap-2 w-full py-1.5 px-2 rounded',
          'hover:bg-muted/50 transition-colors text-left'
        )}
        aria-expanded={isExpanded}
        aria-controls="archived-milestones"
      >
        <ChevronRight
          className={cn(
            'w-4 h-4 transition-transform text-muted-foreground',
            isExpanded && 'rotate-90'
          )}
        />
        <span className="text-sm text-muted-foreground">
          Archived ({archivedNodes.length})
        </span>
      </button>

      <AnimatePresence initial={false}>
        {isExpanded && (
          <motion.div
            id="archived-milestones"
            initial={{ height: 0, opacity: 0 }}
            animate={{ height: 'auto', opacity: 1 }}
            exit={{ height: 0, opacity: 0 }}
            transition={{ duration: 0.2, ease: 'easeInOut' }}
            style={{ overflow: 'hidden' }}
          >
            <div className="mt-1 space-y-1">
              {archivedNodes.map((node) => (
                <GSDTreeNode
                  key={node.id}
                  node={node}
                  depth={0}
                  currentPhaseNumber={0}  // No current phase in archived
                  projectPath={projectPath}
                  isArchived={true}
                />
              ))}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}
```

Key features per CONTEXT.md:
- Collapsed by default (`useState(false)`)
- Shows "Archived (N)" header with chevron
- Dimmed styling via isArchived prop passed to GSDTreeNode
- Smooth expand/collapse animation with Framer Motion
- Archived nodes still clickable to open files

/**
 * GSD state panel - displays milestone/phase/plan tree
 * Integrates GSDTreeView for active milestones and GSDArchivedSection for archived
 */

import { FolderTree } from 'lucide-react';
import { useGSDStore } from '@/stores/gsdStore';
import { GSDTreeView } from './GSDTreeView';
import { GSDArchivedSection } from './GSDArchivedSection';

interface GSDStatePanelProps {
  projectPath: string | null;
}

export function GSDStatePanel({ projectPath }: GSDStatePanelProps) {
  const { treeData, archivedTreeData, isLoading, error } = useGSDStore();

  return (
    <div className="flex flex-col h-full">
      {/* Header */}
      <div className="flex items-center gap-2 px-4 py-3 border-b border-border">
        <FolderTree className="w-5 h-5 text-primary" />
        <h2 className="text-sm font-semibold">State Tree</h2>
      </div>

      {/* Content */}
      <div className="flex-1 overflow-y-auto p-4">
        {isLoading && (
          <div className="text-sm text-muted-foreground">Loading...</div>
        )}

        {error && (
          <div className="text-sm text-destructive">{error}</div>
        )}

        {!isLoading && !error && treeData.length === 0 && (
          <div className="text-sm text-muted-foreground text-center py-8">
            <p>No project data found.</p>
            <p className="mt-1 text-xs">
              Initialize GSD to see your project tree.
            </p>
          </div>
        )}

        {!isLoading && !error && treeData.length > 0 && (
          <>
            {/* Active milestones tree */}
            <GSDTreeView projectPath={projectPath} />

            {/* Archived milestones section */}
            <GSDArchivedSection
              archivedNodes={archivedTreeData}
              projectPath={projectPath}
            />
          </>
        )}
      </div>
    </div>
  );
}
```

1. Add projectPath prop (needed for file paths)
2. Import and use GSDTreeView for active tree
3. Import and use GSDArchivedSection for archived milestones
4. Handle loading, error, and empty states
5. Add header with icon and title
6. Add scrollable content area
